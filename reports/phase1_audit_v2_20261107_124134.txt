"""Audio loading and log-mel features.

This is the ONLY place where feature maths lives. Training, evaluation and the web app must all call
these functions, so the same sound always becomes the same numbers.
"""
import numpy as np
import soundfile as sf
import librosa


def expected_samples(cfg: dict) -> int:
    return int(round(cfg["audio"]["sample_rate"] * cfg["audio"]["expected_duration_s"]))


def expected_frames(cfg: dict) -> int:
    """Number of spectrogram frames for one clip (librosa pads the edges, hence the +1)."""
    return 1 + expected_samples(cfg) // cfg["features"]["hop_length"]


def load_mono_wav(path, cfg: dict) -> np.ndarray:
    """Read a mono WAV as float32 in [-1, 1]. Refuses wrong formats instead of silently fixing them."""
    y, sr = sf.read(str(path), dtype="float32", always_2d=True)
    if sr != cfg["audio"]["sample_rate"]:
        raise ValueError(f"{path}: sample rate is {sr}, expected {cfg['audio']['sample_rate']}")
    if y.shape[1] != 1:
        raise ValueError(f"{path}: has {y.shape[1]} channels, expected 1 (mono)")
    return y[:, 0]


def fix_length(y: np.ndarray, n: int):
    """Pad with zeros or trim to exactly n samples. Returns (signal, 'ok' | 'padded' | 'trimmed')."""
    if len(y) == n:
        return y, "ok"
    if len(y) > n:
        return y[:n], "trimmed"
    return np.pad(y, (0, n - len(y))), "padded"


def log_mel(y: np.ndarray, cfg: dict) -> np.ndarray:
    """Log-mel spectrogram in absolute decibels, shape (n_mels, frames), float32.

    Deliberately NO per-clip scaling: top_db=None and ref=1.0 keep the true loudness, because a louder
    or quieter machine can itself be a sign of a problem."""
    f = cfg["features"]
    mel = librosa.feature.melspectrogram(
        y=y, sr=cfg["audio"]["sample_rate"], n_fft=f["n_fft"], hop_length=f["hop_length"],
        n_mels=f["n_mels"], fmin=f["fmin"], fmax=f["fmax"], power=f["power"], center=True)
    return librosa.power_to_db(mel, ref=1.0, amin=f["amin"], top_db=None).astype(np.float32)


class RunningStats:
    """Mean and standard deviation per mel band, computed file by file (so it fits in 8 GB of RAM)."""

    def __init__(self, n_mels: int):
        self.n_frames = 0
        self.n_files = 0
        self._sum = np.zeros(n_mels, dtype=np.float64)
        self._sumsq = np.zeros(n_mels, dtype=np.float64)

    def update(self, spec: np.ndarray) -> None:
        x = spec.astype(np.float64)
        self.n_frames += x.shape[1]
        self.n_files += 1
        self._sum += x.sum(axis=1)
        self._sumsq += (x ** 2).sum(axis=1)

    def finalize(self, min_std: float = 1e-6):
        if self.n_frames == 0:
            raise ValueError("No data was given to RunningStats")
        mean = self._sum / self.n_frames
        var = np.maximum(self._sumsq / self.n_frames - mean ** 2, 0.0)
        return mean, np.maximum(np.sqrt(var), min_std)


def apply_norm(spec: np.ndarray, mean: np.ndarray, std: np.ndarray) -> np.ndarray:
    """Standardise each mel band with statistics that were fitted on TRAINING data only."""
    return ((spec - mean[:, None]) / std[:, None]).astype(np.float32)


def make_windows(spec: np.ndarray, window_frames: int, hop_frames: int) -> np.ndarray:
    """Cut one spectrogram (n_mels, frames) into overlapping windows -> (n_windows, n_mels, window_frames)."""
    n_frames = spec.shape[1]
    if n_frames < window_frames:
        raise ValueError(f"spectrogram has {n_frames} frames, fewer than the window size {window_frames}")
    starts = range(0, n_frames - window_frames + 1, hop_frames)
    return np.stack([spec[:, s:s + window_frames] for s in starts])

"""Turn every manifest file into a log-mel spectrogram and fit normalisation on TRAINING files only.

    python scripts/extract_features.py            # skips machines that are already done

    norm/classifier.json            mean/std fitted on classifier-TRAINING files (both machine types)
    features_meta.json              settings fingerprint and library versions
"""
import argparse
import json
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

import librosa
import numpy as np
import pandas as pd
import soundfile

from smart_listener import audio
from smart_listener.config import config_fingerprint, load_config, resolve_path


def _save_stats(path: Path, stats: audio.RunningStats, fitted_on: str, file_ids, cfg: dict) -> None:
    mean, std = stats.finalize()
    payload = {"fitted_on": fitted_on, "n_files": stats.n_files, "n_frames": stats.n_frames,
               "n_mels": len(mean), "mean": mean.tolist(), "std": std.tolist(),
               "config_fingerprint": config_fingerprint(cfg),
               "file_ids_sha256": __import__("hashlib").sha256("\n".join(sorted(file_ids)).encode()).hexdigest()}
    path.parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(payload, fh)


def main() -> int:
    ap = argparse.ArgumentParser(description="Extract log-mel features")
    ap.add_argument("--force", action="store_true")
    ap.add_argument("--mono-dir")
    args = ap.parse_args()

    cfg = load_config()
    root = Path(args.mono_dir) if args.mono_dir else resolve_path(cfg, "mono_dir")
    mpath = resolve_path(cfg, "manifest_dir") / "manifest.csv"
    if not mpath.exists():
        print(f"{mpath} not found. Run scripts/build_manifest.py first.")
        return 1
    manifest = pd.read_csv(mpath, keep_default_na=False)
    out = resolve_path(cfg, "features_dir")
    (out / "logmel").mkdir(parents=True, exist_ok=True)

    n_mels, n_frames, n_samples = cfg["features"]["n_mels"], audio.expected_frames(cfg), audio.expected_samples(cfg)
    fingerprint = config_fingerprint(cfg)
    fixed = {"padded": 0, "trimmed": 0}
    t0 = time.time()

    for group, g in manifest.groupby("group", sort=True):
        g = g.sort_values(["label_name", "index_in_label"]).reset_index(drop=True)
        npy = out / "logmel" / f"{group}.npy"
        fcsv = out / "logmel" / f"{group}_files.csv"
        meta_path = out / "features_meta.json"
        cache_valid = False
        if meta_path.exists():
             with open(meta_path, encoding="utf-8") as f:
                  old_meta = json.load(f)
                  cache_valid = old_meta.get("config_fingerprint") == fingerprint
                  if npy.exists() and fcsv.exists() and not args.force and cache_valid:
                      old = pd.read_csv(fcsv)
                      if list(old["file_id"]) == list(g["file_id"]):
                          print(f"[skip] {group}: already done ({len(g)} files)")
                          continue
        arr = np.zeros((len(g), n_mels, n_frames), dtype=np.float32)
        for i, p in enumerate(g["path"]):
            y = audio.load_mono_wav(root / p, cfg)
            y, how = audio.fix_length(y, n_samples)
            if how != "ok":
                fixed[how] += 1
            spec = audio.log_mel(y, cfg)
            if spec.shape != (n_mels, n_frames):
                raise RuntimeError(f"{p}: unexpected spectrogram shape {spec.shape}")
            arr[i] = spec
        if not np.isfinite(arr).all():
            raise RuntimeError(f"{group}: NaN or infinite values in features")
        np.save(npy, arr)
        g[["file_id"]].to_csv(fcsv, index=False)
        print(f"[done] {group}: {len(g)} files, shape {arr.shape}, range {arr.min():.1f} to {arr.max():.1f} dB "
              f"({time.time() - t0:.0f} s)")

    # ---- normalisation statistics: TRAINING files only ----
    for group, g in manifest.groupby("group", sort=True):
        g = g.sort_values(["label_name", "index_in_label"]).reset_index(drop=True)
        arr = np.load(out / "logmel" / f"{group}.npy", mmap_mode="r")
        train = g[g["split_ad"] == "train"]
        assert (train["label"] == 0).all(), "anomaly training split must contain normal files only"
        st = audio.RunningStats(n_mels)
        for i in train.index:
            st.update(arr[i])
        _save_stats(out / "norm" / f"{group}_anomaly.json", st, "split_ad == train", train["file_id"], cfg)
    st = audio.RunningStats(n_mels)
    cls_ids = []
    for group, g in manifest.groupby("group", sort=True):
        g = g.sort_values(["label_name", "index_in_label"]).reset_index(drop=True)
        arr = np.load(out / "logmel" / f"{group}.npy", mmap_mode="r")
        for i in g.index[g["split_cls"] == "train"]:
            st.update(arr[i])
            cls_ids.append(g.loc[i, "file_id"])
    if st.n_files:
        _save_stats(out / "norm" / "classifier.json", st, "split_cls == train", cls_ids, cfg)

    meta = {"created_utc": datetime.now(timezone.utc).isoformat(), "config_fingerprint": fingerprint,
            "features": cfg["features"], "audio": cfg["audio"],
            "versions": {"numpy": np.__version__, "librosa": librosa.__version__, "soundfile": soundfile.__version__},
            "files_padded": fixed["padded"], "files_trimmed": fixed["trimmed"]}
    with open(out / "features_meta.json", "w", encoding="utf-8") as fh:
        json.dump(meta, fh, indent=2)
    print(f"\nFeatures done. Config fingerprint: {fingerprint}. Padded: {fixed['padded']}, trimmed: {fixed['trimmed']}.")
    print(f"Normalisation statistics saved in {out / 'norm'}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Quality checks for the extracted mono WAV files.

EXCLUDED (technical corruption only): unreadable, wrong sample rate, not mono, wrong length, completely
silent, exact duplicates (only the first copy is kept), and label conflicts (same audio, different label).
ONLY WARNED (never excluded): clipping, long silent stretches, unusual loudness. These can be genuine
anomalies, and removing them would make the abnormal test data artificially easy.
"""
import hashlib

import numpy as np
import pandas as pd
import soundfile as sf

from smart_listener import audio


def audit_file(path, cfg: dict) -> dict:
    """Measure one file. Never raises: problems are recorded in the result."""
    q = cfg["quality"]
    rec = {"readable": False, "sr": np.nan, "channels": np.nan, "frames": np.nan, "duration_s": np.nan,
           "peak": np.nan, "rms": np.nan, "dc_offset": np.nan, "clip_fraction": np.nan,
           "silence_fraction": np.nan, "level_db": np.nan, "sha256": "", "error": ""}
    try:
        info = sf.info(str(path))
        rec.update(sr=info.samplerate, channels=info.channels, frames=info.frames,
                   duration_s=round(info.frames / info.samplerate, 4))
        data, _ = sf.read(str(path), dtype="int16", always_2d=True)
    except Exception as exc:  # noqa: BLE001 - any read problem is a finding, not a crash
        rec["error"] = repr(exc)
        return rec
    rec["readable"] = True
    rec["sha256"] = hashlib.sha256(data.tobytes()).hexdigest()
    if data.shape[0] == 0:
        return rec
    x = data[:, 0].astype(np.float32) / 32768.0
    rec["peak"] = float(np.abs(x).max())
    rec["rms"] = float(np.sqrt(np.mean(x.astype(np.float64) ** 2)))
    rec["dc_offset"] = float(x.mean())
    rec["clip_fraction"] = float(np.mean(np.abs(x) >= q["clip_level"]))
    block = max(1, int(cfg["audio"]["sample_rate"] * q["silence_block_ms"] / 1000))
    n_blocks = len(x) // block
    if n_blocks:
        b = x[:n_blocks * block].reshape(n_blocks, block)
        rec["silence_fraction"] = float(np.mean(np.sqrt((b.astype(np.float64) ** 2).mean(axis=1)) < q["silence_block_rms"]))
    if info.samplerate == cfg["audio"]["sample_rate"] and len(x) >= cfg["features"]["n_fft"]:
        rec["level_db"] = float(audio.log_mel(x, cfg).mean())
    return rec


def decide_exclusions(df: pd.DataFrame, cfg: dict) -> pd.DataFrame:
    """Add columns exclude (bool), exclude_reason, and warning flags. df must be in manifest order."""
    a, q = cfg["audio"], cfg["quality"]
    out = df.copy()
    reason = pd.Series("", index=out.index, dtype=object)

    def mark(mask, text):
        nonlocal reason
        mask = mask & (reason == "")
        reason = reason.mask(mask, text)

    mark(~out["readable"].astype(bool), "unreadable")
    mark(out["sr"] != a["sample_rate"], "wrong_sample_rate")
    mark(out["channels"] != 1, "not_mono")
    mark((out["duration_s"] - a["expected_duration_s"]).abs() > a["duration_tolerance_s"], "wrong_duration")
    mark(out["peak"] == 0, "silent")

    ok = out[reason == ""]
    for sha, g in ok[ok["sha256"] != ""].groupby("sha256"):
        if len(g) < 2:
            continue
        if g["label"].nunique() > 1:
            reason.loc[g.index] = "label_conflict"
        else:
            keep = g.index[0]
            reason.loc[g.index[1:]] = f"duplicate_of:{out.loc[keep, 'file_id']}"

    out["exclude_reason"] = reason
    out["exclude"] = reason != ""
    out["warn_clipping"] = out["clip_fraction"] > q["clip_fraction_warn"]
    out["warn_silence"] = out["silence_fraction"] > q["silence_fraction_warn"]
    return out


def drift_table(df: pd.DataFrame, cfg: dict) -> pd.DataFrame:
    """Does loudness change with the file number? A strong trend means neighbouring recordings are
    similar, which is exactly why we split in blocks with gaps."""
    from scipy.stats import spearmanr

    q = cfg["quality"]
    rows = []
    kept = df[~df["exclude"] & df["level_db"].notna()]
    for (mtype, mid, label), g in kept.groupby(["machine_type", "machine_id", "label_name"]):
        if len(g) < q["drift_min_files"]:
            continue
        rho, p = spearmanr(g["index_in_label"], g["level_db"])
        rows.append({"machine_type": mtype, "machine_id": mid, "label_name": label, "n_files": len(g),
                     "rho": float(rho), "p_value": float(p), "strong_trend": bool(abs(rho) >= q["drift_rho_warn"])})
    return pd.DataFrame(rows)

def project_root() -> Path:

def _is_number(v) -> bool:
    return isinstance(v, (int, float)) and not isinstance(v, bool)


def _pos_int(section: dict, key: str, where: str) -> None:
    v = section.get(key)
    if not isinstance(v, int) or isinstance(v, bool) or v <= 0:
        raise ValueError(f"{where}.{key} must be a positive whole number, got {v!r}")


def _pos_num(section: dict, key: str, where: str) -> None:
    v = section.get(key)
    if not _is_number(v) or v <= 0:
        raise ValueError(f"{where}.{key} must be a positive number, got {v!r}")


def validate_config(cfg: dict) -> None:
    """Raise ValueError with a clear message if the configuration is unusable."""
    for name in SECTIONS:
        if not isinstance(cfg.get(name), dict):
            raise ValueError(f"Missing section '{name}' in the configuration file")

    a, f, q, s, d = cfg["audio"], cfg["features"], cfg["quality"], cfg["split"], cfg["dataset"]

    _pos_int(a, "sample_rate", "audio")
    if not isinstance(a.get("channel_index"), int) or a["channel_index"] < 0:
        raise ValueError("audio.channel_index must be 0 or a positive whole number")
    _pos_num(a, "expected_duration_s", "audio")
    if not _is_number(a.get("duration_tolerance_s")) or a["duration_tolerance_s"] < 0:
        raise ValueError("audio.duration_tolerance_s must be 0 or a positive number")

    for key in ("n_fft", "hop_length", "n_mels", "window_frames", "window_hop_frames"):
        _pos_int(f, key, "features")
    if f["hop_length"] > f["n_fft"]:
        raise ValueError("features.hop_length must not be larger than features.n_fft")
    if f["window_hop_frames"] > f["window_frames"]:
        raise ValueError("features.window_hop_frames must not be larger than features.window_frames")
    for key in ("power", "amin"):
        _pos_num(f, key, "features")
    if not _is_number(f.get("fmin")) or f["fmin"] < 0:
        raise ValueError("features.fmin must be 0 or a positive number")
    if not _is_number(f.get("fmax")) or f["fmax"] <= f["fmin"] or f["fmax"] > a["sample_rate"] / 2:
        raise ValueError("features.fmax must be above fmin and not above half the sample rate")
    if f.get("normalization") != "global_per_mel_band":
        raise ValueError("features.normalization must be 'global_per_mel_band'")

    for key in ("clip_level", "clip_fraction_warn", "silence_block_rms", "silence_fraction_warn", "drift_rho_warn"):
        _pos_num(q, key, "quality")
    _pos_int(q, "silence_block_ms", "quality")
    _pos_int(q, "drift_min_files", "quality")

    _pos_int(s, "seed", "split")
    an = s.get("anomaly")
    if not isinstance(an, dict):
        raise ValueError("Missing section 'split.anomaly'")
    if an.get("strategy") not in ("interleaved", "block", "random"):
        raise ValueError("split.anomaly.strategy must be 'interleaved', 'block' or 'random'")
    if not _is_number(an.get("abnormal_val_fraction")) or not 0 < an["abnormal_val_fraction"] < 1:
        raise ValueError("split.anomaly.abnormal_val_fraction must be between 0 and 1")
    if not isinstance(an.get("buffer_files"), int) or an["buffer_files"] < 0:
        raise ValueError("split.anomaly.buffer_files must be 0 or a positive whole number")
    _pos_int(an, "min_train_files", "split.anomaly")

    cl = s.get("classifier")
    if not isinstance(cl, dict):
        raise ValueError("Missing section 'split.classifier'")
    seen = []
    for key in ("train_ids", "val_ids", "test_ids"):
        ids = cl.get(key)
        if not isinstance(ids, list) or not ids:
            raise ValueError(f"split.classifier.{key} must be a non-empty list")
        seen += ids
    if len(seen) != len(set(seen)):
        raise ValueError("split.classifier: the same machine id appears in more than one split")
    unknown = set(seen) - set(d.get("machine_ids", []))
    if unknown:
        raise ValueError(f"split.classifier uses ids that are not in dataset.machine_ids: {sorted(unknown)}")

    for key in ("mono_dir", "manifest_dir", "features_dir", "figures_dir"):
        if not isinstance(cfg["paths"].get(key), str) or not cfg["paths"][key]:
            raise ValueError(f"paths.{key} must be a folder path (text)")


def load_config(path=None) -> dict:
    """Read the YAML configuration. Uses config/audio_config.yaml when no path is given."""
    path = Path(path) if path else project_root() / "config" / "audio_config.yaml"

def config_fingerprint(cfg: dict) -> str:
    """Short code that changes whenever any audio or feature setting changes.

    Saved next to every feature file. Training and the app compare it, so features made with
    different settings can never be mixed by accident."""
    payload = {"audio": cfg["audio"], "features": cfg["features"]}
    text = json.dumps(payload, sort_keys=True, separators=(",", ":"))
    return hashlib.sha256(text.encode("utf-8")).hexdigest()[:16]

"""Dataset manifest: one row per audio file, with its label and its split.

Labels come ONLY from the folder name (normal / abnormal). Splits are assigned here, on whole files,
BEFORE any windowing, so overlapping windows can never leak between train, validation and test.
"""
import re
import zlib
from pathlib import Path

import numpy as np
import pandas as pd

LABELS = {"normal": 0, "abnormal": 1}


def natural_key(text: str):
    """Sort 'file2' before 'file10' (and keep zero-padded names in order)."""
    return [int(t) if t.isdigit() else t.lower() for t in re.split(r"(\d+)", text)]


# ----------------------------------------------------------------------------- scanning
def scan_folder(root, cfg: dict):
    """Find every WAV under root/<type>/<id>/<label>/. Returns (DataFrame, list of unexpected paths)."""
    root = Path(root)
    types = set(cfg["dataset"]["machine_types"])
    ids = set(cfg["dataset"]["machine_ids"])
    rows, unexpected = [], []
    for p in sorted(root.rglob("*")):
        if p.is_dir():
            continue
        parts = p.relative_to(root).parts
        if p.suffix.lower() != ".wav" or len(parts) != 4 or parts[0] not in types \
                or parts[1] not in ids or parts[2] not in LABELS:
            if p.suffix.lower() != ".csv":           # inventory CSVs are expected
                unexpected.append(p.relative_to(root).as_posix())
            continue
        mtype, mid, label_name, fname = parts
        rows.append({"file_id": f"{mtype}_{mid}_{label_name}_{Path(fname).stem}",
                     "path": p.relative_to(root).as_posix(),
                     "machine_type": mtype, "machine_id": mid, "group": f"{mtype}_{mid}",
                     "label_name": label_name, "label": LABELS[label_name],
                     "snr_db": cfg["dataset"]["snr_db"]})
    df = pd.DataFrame(rows)
    if df.empty:
        return df, unexpected
    # natural order of the file names inside each machine and label (this order defines the "blocks")
    df["_k"] = df["path"].map(lambda s: tuple(natural_key(Path(s).name)))
    df = df.sort_values(["machine_type", "machine_id", "label_name", "_k"], kind="stable")
    df = df.drop(columns="_k").reset_index(drop=True)
    df["index_in_label"] = df.groupby(["machine_type", "machine_id", "label_name"]).cumcount()
    return df, unexpected


# ----------------------------------------------------------------------------- anomaly-detector splits
def _group_seed(seed: int, group: str) -> int:
    return (seed + zlib.crc32(group.encode("utf-8"))) % (2 ** 32)


def _block_split(normal_pos, abn_pos, an_cfg, group):
    """Return dict split -> list of row positions for ONE machine. Layout for normal files:
    [validation block][gap][ ... training ... ][gap][test block]. Abnormal: [validation][gap][test]."""
    b = an_cfg["buffer_files"]
    frac = an_cfg["abnormal_val_fraction"]
    a_n, n_n = len(abn_pos), len(normal_pos)
    v = int((a_n - b) * frac)
    t = a_n - b - v
    if v < 1 or t < 1:
        raise ValueError(f"{group}: only {a_n} abnormal files, too few to make validation and test sets")
    n_train = n_n - v - t - 2 * b
    if n_train < an_cfg["min_train_files"]:
        raise ValueError(f"{group}: only {n_train} normal files would remain for training "
                         f"(minimum {an_cfg['min_train_files']})")
    return {
        "val": list(abn_pos[:v]) + list(normal_pos[:v]),
        "test": list(abn_pos[a_n - t:]) + list(normal_pos[n_n - t:]),
        "train": list(normal_pos[v + b:n_n - t - b]),
        "buffer": list(abn_pos[v:v + b]) + list(normal_pos[v:v + b]) + list(normal_pos[n_n - t - b:n_n - t]),
    }


def _interleaved_split(normal_pos, abn_pos, an_cfg, group):
    """Spread validation and test over the WHOLE recording, so train, validation and test see the same
    sessions, loudness and fault types (a single block at each end would not).

    The abnormal sequence is cut into K chunks (K a multiple of 4). Chunks go to validation, test, test,
    validation, validation, test, test, ... so both sets reach early and late recordings. The first
    `buffer` files of a chunk are dropped whenever the previous chunk belongs to the other set. For each chunk, the SAME relative stretch of the
    normal sequence supplies exactly as many normal files, centred in the chunk with a gap on both sides; all other normal files
    are training files. So validation and test always have equal normal and abnormal counts."""
    b = an_cfg["buffer_files"]
    a_n, n_n = len(abn_pos), len(normal_pos)
    k = int(np.clip((a_n // 15) // 4 * 4, 4, an_cfg.get("interleaved_max_blocks", 16)))
    a_chunks = np.array_split(np.arange(a_n), k)
    n_chunks = np.array_split(np.arange(n_n), k)
    parts = {"val": [], "test": [], "train": [], "buffer": []}
    for j, (ac, nc) in enumerate(zip(a_chunks, n_chunks)):
        name = "val" if j % 4 in (0, 3) else "test"
        prev = "val" if (j - 1) % 4 in (0, 3) else "test"
        keep_a = ac[b:] if (j > 0 and prev != name) else ac   # gap only where val meets test
        parts["buffer"] += [abn_pos[i] for i in ac[:len(ac) - len(keep_a)]]
        parts[name] += [abn_pos[i] for i in keep_a]
        s = len(keep_a)
        if s + 2 * b > len(nc):
            raise ValueError(f"{group}: normal chunk {j} too short for {s} files plus gaps")
        pre = (len(nc) - s - 2 * b) // 2             # centre the evaluation files inside the chunk
        parts["train"] += [normal_pos[i] for i in nc[:pre]]
        parts["buffer"] += [normal_pos[i] for i in nc[pre:pre + b]]
        parts[name] += [normal_pos[i] for i in nc[pre + b:pre + b + s]]
        parts["buffer"] += [normal_pos[i] for i in nc[pre + b + s:pre + 2 * b + s]]
        parts["train"] += [normal_pos[i] for i in nc[pre + 2 * b + s:]]
    if min(len(parts["val"]), len(parts["test"])) // 2 < 5 or len(parts["train"]) < an_cfg["min_train_files"]:
        raise ValueError(f"{group}: too few files after interleaved split "
                         f"(train {len(parts['train'])}, val {len(parts['val'])}, test {len(parts['test'])})")
    return parts


def _random_split(normal_pos, abn_pos, an_cfg, group, seed):
    rng = np.random.default_rng(_group_seed(seed, group))
    a_n, n_n = len(abn_pos), len(normal_pos)
    v = int(a_n * an_cfg["abnormal_val_fraction"])
    t = a_n - v
    if v < 1 or t < 1:
        raise ValueError(f"{group}: only {a_n} abnormal files, too few to make validation and test sets")
    if n_n - v - t < an_cfg["min_train_files"]:
        raise ValueError(f"{group}: too few normal files left for training")
    a = rng.permutation(np.array(abn_pos))
    n = rng.permutation(np.array(normal_pos))
    return {"val": list(a[:v]) + list(n[:v]), "test": list(a[v:]) + list(n[v:v + t]),
            "train": list(n[v + t:]), "buffer": []}


def assign_anomaly_splits(df: pd.DataFrame, cfg: dict) -> pd.DataFrame:
    """Add column split_ad (train / val / test / buffer) for the normal-vs-abnormal detector.

    train = normal files only. val and test each contain as many normal as abnormal files."""
    out = df.copy()
    out["split_ad"] = "unused"
    an = cfg["split"]["anomaly"]
    for group, g in out.groupby("group"):
        normal = list(g[g["label"] == 0].sort_values("index_in_label").index)
        abn = list(g[g["label"] == 1].sort_values("index_in_label").index)
        if an["strategy"] == "interleaved":
            parts = _interleaved_split(normal, abn, an, group)
        elif an["strategy"] == "block":
            parts = _block_split(normal, abn, an, group)
        else:
            parts = _random_split(normal, abn, an, group, cfg["split"]["seed"])
        for name, idx in parts.items():
            out.loc[idx, "split_ad"] = name
    return out


# ----------------------------------------------------------------------------- classifier splits
def assign_classifier_splits(df: pd.DataFrame, cfg: dict) -> pd.DataFrame:
    """Add column split_cls for fan-vs-pump: whole machines go to ONE split only."""
    cl = cfg["split"]["classifier"]
    mapping = {i: "train" for i in cl["train_ids"]}
    mapping.update({i: "val" for i in cl["val_ids"]})
    mapping.update({i: "test" for i in cl["test_ids"]})
    out = df.copy()
    out["split_cls"] = out["machine_id"].map(mapping).fillna("unused")
    for split in ("train", "val", "test"):
        present = set(out.loc[out["split_cls"] == split, "machine_type"])
        missing = set(cfg["dataset"]["machine_types"]) - present
        if missing:
            raise ValueError(f"classifier split '{split}' has no files for {sorted(missing)}")
    return out
